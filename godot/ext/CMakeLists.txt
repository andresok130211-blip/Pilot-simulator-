cmake_minimum_required(VERSION 3.13)
project(jsvm LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
# godot-cpp и quickjs клонируются рядом (tools/build_ext.sh)
set(GODOTCPP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/godot-cpp" CACHE PATH "")
set(QJS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/quickjs" CACHE PATH "")
add_subdirectory(${GODOTCPP_DIR} godot-cpp)
file(READ "${QJS_DIR}/VERSION" QJS_VER)
string(STRIP "${QJS_VER}" QJS_VER)
add_library(quickjs STATIC ${QJS_DIR}/quickjs.c ${QJS_DIR}/cutils.c ${QJS_DIR}/dtoa.c ${QJS_DIR}/libregexp.c ${QJS_DIR}/libunicode.c)
target_include_directories(quickjs PUBLIC ${QJS_DIR})
target_compile_definitions(quickjs PRIVATE _GNU_SOURCE CONFIG_VERSION="${QJS_VER}")
target_compile_options(quickjs PRIVATE -w -funsigned-char -fwrapv -O2)
add_library(jsvm SHARED src/jsvm.cpp src/register_types.cpp)
target_link_libraries(jsvm PRIVATE godot::cpp quickjs)
set(JSVM_SUFFIX "linux.x86_64" CACHE STRING "")
set_target_properties(jsvm PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/../bin OUTPUT_NAME "jsvm.${JSVM_SUFFIX}")
